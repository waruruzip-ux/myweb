function weatherCodeToInfo(code){
  const map = {

    61:['🌧️','약한 비'],63:['🌧️','비'],65:['🌧️','강한 비'],
    71:['🌨️','약한 눈'],73:['🌨️','눈'],75:['❄️','강한 눈'],
    80:['🌦️','소나기'],81:['🌧️','강한 소나기'],82:['⛈️','매우 강한 소나기'],

  };

async function loadWeather(){
  // 충남 천안시 서북구 성정동 고정 좌표
  const lat = 36.8140, lon = 127.1090;
  try{
    const res = await fetch(`https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current_weather=true&timezone=Asia%2FSeoul`);
    const data = await res.json();
    const cw = data.current_weather;
    const [emoji, label] = weatherCodeToInfo(cw.weathercode);
    weatherEmojiEl.textContent = emoji;
    weatherTextEl.textContent = `${Math.round(cw.temperature)}°C · ${label}`;
  }catch(e){
    weatherTextEl.textContent = '날씨 정보를 가져오지 못했어요';
  }
}
loadWeather();

function updateCountdown(){
  const now = new Date();
  const nowMs = now.getTime();

  const lunch = new Date(now); lunch.setHours(12,0,0,0);
  const leave = new Date(now); leave.setHours(17,30,0,0);
  const midnight = new Date(now); midnight.setHours(24,0,0,0);
  const dayStart = new Date(now); dayStart.setHours(9,0,0,0);

  if(nowMs < lunch.getTime()){
    const remainMs = lunch.getTime() - nowMs;
    renderTime('🍚 점심시간까지 남은 시간', remainMs, '점심시간 12:00');
    const total = lunch.getTime() - dayStart.getTime();
    setFaceByRatio(remainMs/total);
    return;
  }
  if(nowMs < leave.getTime()){
    const remainMs = leave.getTime() - nowMs;
    renderTime('🏃 퇴근까지 남은 시간', remainMs, '퇴근시간 17:30');
    const total = leave.getTime() - lunch.getTime();
    setFaceByRatio(remainMs/total);
    return;
  }
  // 17:30 ~ 24:00 : 토닥토닥 위로 시간
  const remainMs = midnight.getTime() - nowMs;
  countdownLabel.textContent = '🐾 오늘도 고생했어, 토닥토닥';
  countdownTime.textContent = fmt(remainMs);
  countdownSub.textContent = '하루 마무리까지 · 폭 쉬어요';
  setFace(4);
}
function fmt(ms){
  const h = Math.floor(ms/3600000);
  const m = Math.floor((ms%3600000)/60000);
  const s = Math.floor((ms%60000)/1000);
  return `${pad(h)}:${pad(m)}:${pad(s)}`;
}
function renderTime(label, remainMs, sub){
  countdownLabel.textContent = label;
  countdownTime.textContent = fmt(remainMs);
  countdownSub.textContent = sub;
}
function setFaceByRatio(remainRatio){
  remainRatio = Math.max(0, Math.min(1, remainRatio));
  let idx;
  if(remainRatio > 0.66) idx = 0;
  else if(remainRatio > 0.4) idx = 1;
  else if(remainRatio > 0.15) idx = 2;
  else idx = 3;
  setFace(idx);
}

/* ---------------- 할일 기록 (window.storage) ---------------- */
const todoInput = document.getElementById('todoInput');
const todoAddBtn = document.getElementById('todoAddBtn');
const todoListEl = document.getElementById('todoList');
const tabBtns = [...document.querySelectorAll('.tab-btn')];
let currentTab = 'today';
let todosCache = { today: [], week: [] };

async function loadTodos(tab){
  try{
    const res = await window.storage.get(`todos:${tab}`, false);
    todosCache[tab] = res ? JSON.parse(res.value) : [];
  }catch(e){
    todosCache[tab] = [];
  }
}
async function saveTodos(tab){
  try{
    await window.storage.set(`todos:${tab}`, JSON.stringify(todosCache[tab]), false);
  }catch(e){ console.error('저장 실패', e); }
}
function renderTodos(){
  const list = todosCache[currentTab];
  todoListEl.innerHTML = '';
  if(!list.length){
    todoListEl.innerHTML = '<div class="empty-msg">아직 등록된 할일이 없어요 ✏️</div>';
    return;
  }
  list.forEach((item, i)=>{
    const li = document.createElement('li');
    if(item.done) li.classList.add('done');
    li.innerHTML = `
      <input type="checkbox" ${item.done?'checked':''} data-i="${i}">
      <span class="txt">${item.text}</span>
      <button class="del" data-i="${i}">✕</button>
    `;
    todoListEl.appendChild(li);
  });
}
todoListEl.addEventListener('click', async (e)=>{
  const i = e.target.dataset.i;
  if(i === undefined) return;
  if(e.target.matches('input[type=checkbox]')){
    todosCache[currentTab][i].done = e.target.checked;
    renderTodos();
    await saveTodos(currentTab);
  } else if(e.target.matches('button.del')){
    todosCache[currentTab].splice(i,1);
    renderTodos();
    await saveTodos(currentTab);
  }
});
async function addTodo(){
  const val = todoInput.value.trim();
  if(!val) return;
  todosCache[currentTab].push({text:val, done:false});
  todoInput.value = '';
  renderTodos();
  await saveTodos(currentTab);
}
todoAddBtn.addEventListener('click', addTodo);
todoInput.addEventListener('keydown', e=>{ if(e.key==='Enter') addTodo(); });
tabBtns.forEach(btn=>{
  btn.addEventListener('click', async ()=>{
    tabBtns.forEach(b=>b.classList.remove('active'));
    btn.classList.add('active');
    currentTab = btn.dataset.tab;
    if(!todosCache[currentTab] || todosCache[currentTab].length===0){
      await loadTodos(currentTab);
    }
    renderTodos();
  });
});

function appendMsg(text, who){
  const div = document.createElement('div');
  div.className = `msg ${who}`;
  div.textContent = text;
  chatBox.appendChild(div);
  chatBox.scrollTop = chatBox.scrollHeight;

}
